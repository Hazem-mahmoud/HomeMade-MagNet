40class TemporalBlock(nn.Module):
41 def __init__(
42 self,
43 n_inputs,
44 n_outputs,
45 kernel_size,
46 stride,
47 dilation,
48 residual=True,
49 double_layered=True,
50 dropout=0.0,
51 act_func=None,
52 ):
53 super(TemporalBlock, self).__init__()
54 padding = ((kernel_size - 1) // 2) * dilation
55
56 self.conv1 = weight_norm(
57 nn.Conv1d(
58 n_inputs,
59 n_outputs,
60 kernel_size,
61 stride=stride,
62 padding=padding,
63 dilation=dilation,
64 padding_mode="circular",
65 )
66 )
67 self.relu1 = ACTIVATION_FUNCS.get(act_func, nn.Identity)()
68 self.dropout1 = nn.Dropout1d(dropout)
69 if double_layered:
70 self.relu2 = nn.Identity()
71 self.conv2 = weight_norm(
72 nn.Conv1d(
73 n_inputs,
74 n_outputs,
75 kernel_size,
76 stride=stride,
77 padding=padding,
78 dilation=dilation,
79 padding_mode="circular",
80 )
81 )
82 self.dropout2 = nn.Dropout1d(dropout)
83 self.net = nn.Sequential(
84 self.conv1,
85 self.relu1,
86 self.dropout1,
87 self.conv2,
88 self.relu2,
89 self.dropout2,
90 )
91 else:
92 self.net = nn.Sequential(self.conv1, self.relu1, self.dropout1)
93 self.relu = nn.ReLU()
94 self.residual = residual
95 if residual:
96 self.downsample = (
97 nn.Conv1d(n_inputs, n_outputs, 1) if n_inputs != n_outputs else None
98 )
99 else:
100 self.downsample = None
101 self.double_layered = double_layered
102 self.init_weights()
103
104 def init_weights(self):
105 self.conv1.weight.data.normal_(0, 0.01)
106 if self.double_layered:
107 self.conv2.weight.data.normal_(0, 0.01)
108 if self.downsample is not None:
109 self.downsample.weight.data.normal_(0, 0.01)
110
111 def forward(self, x):
112 out = self.net(x)
113 if self.residual:
114 res = x if self.downsample is None else self.downsample(x)
115 y = torch.clip(
116 out + res, -10, 10
117 )
118 y = self.relu(y)
119 else:
120 y = out
121 return y

124class TCNWithScalarsAsBias(nn.Module):
125 def __init__(
126 self,
127 num_input_scalars,
128 num_input_ts=1,
129 tcn_layer_cfg=None,
130 scalar_layer_cfg=None,
131 ):
132 super().__init__()
133 self.num_input_ts = num_input_ts
134 self.num_input_scalar = num_input_scalars
135 tcn_layer_cfg = tcn_layer_cfg or {
136 "f": [
137 {"units": (num_input_scalars + 1), "act_func": "tanh"},
138 {"units": 8, "act_func": "tanh"},
139 {"units": 1},
140 ]
141 }
142 scalar_layer_cfg = scalar_layer_cfg or {
143 "f": [
144 {"units": num_input_scalars, "act_func": "tanh"},
145 ]
146 }
147 # build CNN layer path
148 cnn_layers = []
149 dilation_offset = tcn_layer_cfg.get("starting_dilation_rate", 2) # >= 0
150 for i, l_cfg in enumerate(tcn_layer_cfg["f"]):
151 kernel_size = l_cfg.get("kernel_size", 9)
152 dropout_rate = tcn_layer_cfg.get("dropout", 0.0)
153 dilation_size = 2 ** (i + dilation_offset)
154 if i == 0:
155 in_channels = num_input_ts
156 else:
157 in_channels = tcn_layer_cfg["f"][i - 1]["units"]
158 cnn_layers += [
159 TemporalBlock(
160 in_channels,
161 l_cfg["units"],
162 kernel_size,
163 stride=1,
164 dilation=dilation_size,
165 residual=tcn_layer_cfg.get("residual", False),
166 double_layered=tcn_layer_cfg.get("double_layered", False),
167 dropout=dropout_rate,
168 act_func=l_cfg.get("act_func", nn.Identity),
169 ),
170 ]
171 if i == 0:
172 self.ts_branch = cnn_layers.pop()
173 self.upper_tcn = nn.Sequential(*cnn_layers)
174 # build scalar NN path
175 scalar_layers = []
176 fan_in = num_input_scalars
177 for i, l_cfg in enumerate(scalar_layer_cfg["f"]):
178 scalar_layers.append(nn.Linear(fan_in, l_cfg["units"]))
179 scalar_layers.append(ACTIVATION_FUNCS.get(l_cfg["act_func"], nn.Identity)())
180 fan_in = l_cfg["units"]
181
182 self.scalar_branch = nn.Sequential(*scalar_layers)
183
184 def forward(self, x_ts, x_scalars):
185 """x_ts has shape (#batch, #channels, #length)"""
186 b_proc = self.ts_branch(x_ts)
187 scalar_proc = self.scalar_branch(x_scalars)
188 catted = torch.cat(
189 [
190 b_proc[:, : -scalar_proc.shape[1], :],
191 b_proc[:, -scalar_proc.shape[1] :, :] + scalar_proc.unsqueeze(-1),
192 ],
193 dim=1,
194 )
195 y = self.upper_tcn(catted)
196 y = y + x_ts[:, [0], :]
197 y = y - y.mean(dim=-1).unsqueeze(-1)
198 return y

200class LossPredictor(nn.Module):
201 def __init__(self, h_predictor):
202 super().__init__()
203 self.h_predictor = h_predictor
204 self.post_processor = nn.Sequential(
205 nn.Linear(self.h_predictor.num_input_scalar, 8),
206 nn.Tanh(),
207 nn.Linear(8, 1),
208 nn.Tanh()
209 )
210
211 def forward(self, x_ts, x_scalars, stats):
212 # 1. Predict H-field (Normalized)
213 h_pred = self.h_predictor(x_ts, x_scalars).permute(2, 0, 1) # (Seq, Batch, 1)
214
215 # 2. De-normalize Physical Quantities for the Shoelace Formula
216 # We need real units: B in Tesla, H in A/m, Freq in Hz
217
218 # Recover Frequency (Hz)
219 # x_scalars[:, 0] is Standardized Freq: (F - mu) / sigma
220 f_norm = x_scalars[:, 0]
221 freq_hz = (f_norm * stats['freq_std']) + stats['freq_mean']
222 freq_hz = torch.abs(freq_hz) # Ensure positive
223
224 # Recover B (Tesla)
225 # Scaled B and H (using learnable/fixed limits to ensure valid range)
226 # Note: Paderborn hardcoded these offsets (+5) to ensure positivity for log.
227 b_lim = 0.5 # Approx max Tesla
228 h_lim = 100.0 # Approx max A/m
229
230 # x_ts is (Batch, Channels, Seq) = (Batch, 1, Seq)
231 # We want (Seq, Batch, 1)
232 scaled_b = x_ts.permute(2, 0, 1) * b_lim
233 scaled_h = h_pred * h_lim
234
235 # Shoelace
236 term1 = scaled_b
237 term2 = torch.roll(scaled_h, 1, dims=0) - torch.roll(scaled_h, -1, dims=0)
238 area = 0.5 * torch.sum(term1 * term2, dim=0) # (Batch, 1)
239
240 # 4. Compute Power Loss: Pv = Freq * Area * (Correction Factor)
241 # Correction factor allows network to adjust for non-ideal shapes
242 correction = 1.0 + 0.1 * self.post_processor(x_scalars)
243
244 ploss_pred = freq_hz.unsqueeze(1) * torch.abs(area) * correction
245
246 # Return Log(Pv) to match Log-Loss Target
247 return torch.log(ploss_pred + 1e-6)
