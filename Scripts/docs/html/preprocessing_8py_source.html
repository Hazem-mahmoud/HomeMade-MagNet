9def normalize_data(data, method='standard', stats=None):
10 """
11 Normalizes data using Global Scaling (Statistics computed over ALL data).
12
13 Args:
14 data (np.array): Input data.
15 method (str): 'standard' (Z-score), 'minmax', or 'none'.
16 stats (dict, optional): Pre-computed stats.
17
18 Returns:
19 norm_data (np.array): Normalized data.
20 stats (dict): Statistics used.
21 """
22 # Force float32
23 data = data.astype(np.float32)
24
25 if method == 'none':
26 return data, {}
27
28 if method == 'standard':
29 if stats:
30 mean = stats['mean']
31 std = stats['std']
32 else:
33 mean = np.mean(data, axis=None)
34 std = np.std(data, axis=None)
35 if std == 0: std = 1.0
36 stats = {'mean': mean, 'std': std}
37
38 norm_data = (data - mean) / std
39 return norm_data, stats
40
41 elif method == 'minmax':
42 if stats:
43 min_val = stats['min']
44 max_val = stats['max']
45 else:
46 min_val = np.min(data, axis=None)
47 max_val = np.max(data, axis=None)
48 stats = {'min': min_val, 'max': max_val}
49
50 denom = max_val - min_val
51 if denom == 0: denom = 1.0
52
53 norm_data = (data - min_val) / denom
54 return norm_data, stats
55
56 elif method == 'log10':
57 # Log10 normalization
58 norm_data = np.log10(np.abs(data) + 1e-6)
59 return norm_data, {}
60
61 else:
62 raise ValueError(f"Unknown method: {method}")

65def prepare_dataset(features, targets, test_ratio=0.2, norm_config=None):
66 """
67 Generic function to split and normalize ANY set of features and targets.
68
69 Args:
70 features (dict): Dictionary mapping FeatureName to data array.
71 targets (dict): Dictionary mapping TargetName to data array.
72 test_ratio (float): Fraction of data to use for testing.
73 norm_config (dict): Dictionary mapping feature/target names to methods.
74 e.g. {'B': 'standard', 'Loss': 'log10'}.
75 Defaults to 'standard' for features and 'none' for targets if not specified.
76
77 Returns:
78 data_split (dict): Contains 'train' and 'test' dictionaries.
79
80 - `data_split['train']['inputs']['FeatureName']`
81 - `data_split['train']['targets']['TargetName']`
82
83 stats (dict): The statistics used for normalization for each feature/target.
84 """
85 if norm_config is None:
86 norm_config = {}
87
88 # 1. Determine Split Index
89 # Assume all arrays have same length N in dimension 0.
90 any_key = next(iter(features))
91 N = len(features[any_key])
92 split_idx = int(N * (1 - test_ratio))
93
94 split_data = {
95 'train': {'inputs': {}, 'targets': {}},
96 'test': {'inputs': {}, 'targets': {}}
97 }
98 stats_out = {}
99
100 print(f"Splitting Dataset: {N} Samples -> {split_idx} Train, {N - split_idx} Test")
101
102 # 2. Process Inputs (Features)
103 print("Processing Features...")
104 for name, data in features.items():
105 # Get method (Default to 'standard' for inputs)
106 method = norm_config.get(name, 'standard')
107
108 # Split
109 train_part = data[:split_idx]
110 test_part = data[split_idx:]
111
112 # Normalize Train
113 print(f" - Normalizing '{name}' with {method}...")
114 train_norm, stat = normalize_data(train_part, method=method)
115 stats_out[name] = stat
116
117 # Normalize Test (using Train stats)
118 test_norm, _ = normalize_data(test_part, method=method, stats=stat)
119
120 split_data['train']['inputs'][name] = train_norm
121 split_data['test']['inputs'][name] = test_norm
122
123 # 3. Process Targets
124 print("Processing Targets...")
125 for name, data in targets.items():
126 # Get method (Default to 'none' for targets unless specified)
127 # Often we don't normalize targets for regression unless specifically asked (like Log Loss)
128 method = norm_config.get(name, 'none')
129
130 # Split
131 train_part = data[:split_idx]
132 test_part = data[split_idx:]
133
134 # Normalize Train
135 if method != 'none':
136 print(f" - Normalizing Target '{name}' with {method}...")
137
138 train_norm, stat = normalize_data(train_part, method=method)
139 if method != 'none':
140 stats_out[name] = stat
141
142 # Normalize Test
143 test_norm, _ = normalize_data(test_part, method=method, stats=stat)
144
145 split_data['train']['targets'][name] = train_norm
146 split_data['test']['targets'][name] = test_norm
147
148 return split_data, stats_out

154def calculate_flux_density(voltage, sampling_time, secondary_turns, effective_area):
155 """ Calculates B (Flux Density). """
156 v_mean = np.mean(voltage, axis=1, keepdims=True)
157 v_clean = voltage - v_mean
158 flux = cumulative_trapezoid(v_clean, axis=-1, initial=0)
159
160 if np.ndim(sampling_time) == 0:
161 flux = flux * sampling_time
162 else:
163 st = np.array(sampling_time).reshape(-1, 1)
164 flux = flux * st
165
166 b = flux / (secondary_turns * effective_area)
167 b_mean = np.mean(b, axis=1, keepdims=True)
168 b = b - b_mean
169
170 return b.astype(np.float32)

172def calculate_magnetizing_force(current, primary_turns, effective_length):
173 """ Calculates H (Magnetizing Force). """
174 h = (primary_turns * current) / effective_length
175 return h.astype(np.float32)
176
177def calculate_volumetric_loss(b_field, h_field, frequency):
178 """ Calculates Volumetric Power Loss (Target). """
179 energy_density = trapezoid(y=h_field, x=b_field, axis=-1)
180 energy_density = np.abs(energy_density)
181
182 if np.ndim(frequency) > 0:
183 freq = np.array(frequency).squeeze()
184 else:
185 freq = frequency
186
187 pv = energy_density * freq
188 return pv.astype(np.float32)

198def process_magnet_dataset(voltage, current, frequency, mag_props, dt, model_type='scaler', config_path=None, extra_features=None):
199 """
200 Orchestrates the physics calculation -> Feature Assembly -> Normalization.
201
202 Args:
203 voltage, current: Raw arrays
204 frequency: Raw frequency array
205 mag_props, dt: Constants
206 model_type (str): Key in config['models'] (e.g. 'scaler', 'cnn')
207 config_path (str, optional): Path to config.yaml.
208 extra_features (dict, optional): Additional raw features (e.g. {'Temperature': ..., 'Hdc': ...})
209 """
210 norm_config = {}
211
212 # 1. Load Config if path provided
213 if config_path:
214 full_config = load_config(config_path)
215 if 'models' in full_config and model_type in full_config['models']:
216 model_conf = full_config['models'][model_type]
217 if 'features' in model_conf:
218 # Merge inputs and targets into one simple config dict for prepare_dataset
219 # prepare_dataset expects {'B': 'method', 'Loss': 'method'}
220 feats = model_conf['features'].get('inputs', {})
221 targs = model_conf['features'].get('targets', {})
222 norm_config = {**feats, **targs}
223 print(f"Loaded config for '{model_type}': {norm_config}")
224 else:
225 print(f"WARNING: No 'features' section found for model '{model_type}'. Using defaults.")
226 else:
227 print(f"WARNING: Model '{model_type}' not found in config. Using defaults.")
228
229 # Fallback default if empty
230 if not norm_config:
231 # Added defaults for Temp/Hdc just in case
232 norm_config = {'B': 'standard', 'H': 'standard', 'Loss': 'standard', 'Frequency': 'standard',
233 'Temperature': 'standard', 'Hdc': 'standard', 'Duty': 'standard'}
234
235 print("--- 1. Physics Calculations ---")
236 B_raw = calculate_flux_density(voltage, dt, mag_props['N_sec'], mag_props['Ae'])
237 H_raw = calculate_magnetizing_force(current, mag_props['N_prim'], mag_props['Le'])
238 Loss_raw = calculate_volumetric_loss(B_raw, H_raw, frequency)
239
240 # Calculate scalar properties (Peak-to-Peak / 2)
241 B_pk = (np.max(B_raw, axis=1, keepdims=True) - np.min(B_raw, axis=1, keepdims=True)) / 2
242 H_pk = (np.max(H_raw, axis=1, keepdims=True) - np.min(H_raw, axis=1, keepdims=True)) / 2
243
244 # Pack Potential Features
245 all_features = {
246 'B': B_raw,
247 'H': H_raw,
248 'B_pk': B_pk,
249 'H_pk': H_pk,
250 'Frequency': frequency.reshape(-1, 1) if np.ndim(frequency) > 0 else np.full((len(B_raw), 1), frequency)
251 }
252
253 if extra_features:
254 # Reshape scalers if needed?
255 # Assuming extra_features are (N_exp,) or (N_exp, 1) or matching B_raw length
256 # Let's ensure they are at least 2D (N, 1) if they are scalars per experiment
257 for k, v in extra_features.items():
258 if np.ndim(v) == 1:
259 all_features[k] = v.reshape(-1, 1)
260 else:
261 all_features[k] = v
262
263 all_targets = {
264 'Loss': Loss_raw
265 }
266
267 # 2. Filter Features/Targets based on Config
268 # We only include keys that are in the norm_config
269 selected_features = {k: v for k, v in all_features.items() if k in norm_config}
270 selected_targets = {k: v for k, v in all_targets.items() if k in norm_config}
271
272 if not selected_features:
273 print("WARNING: No features selected! Checking config...")
274 # Fallback to keeping all if config was malformed to avoid returning nothing
275 selected_features = all_features
276
277 print(f"Selected Features: {list(selected_features.keys())}")
278 print(f"Selected Targets: {list(selected_targets.keys())}")
279
280 print("--- 2. Generic Normalization & Split ---")
281 data_split, stats = prepare_dataset(selected_features, selected_targets, test_ratio=0.2, norm_config=norm_config)
282
283 return data_split, stats
